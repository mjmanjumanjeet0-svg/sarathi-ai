const CHAT_MODEL = "openai/gpt-oss-20b"; const SYSTEM_PROMPT = ` You are Saarthi AI, a helpful, careful, multilingual assistant. "*" : "null"; return { "Access-Control-Allow-Origin": allowedOrigin, "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Access-Control-Max-Age": "86400", "Vary": "Origin" }; } function jsonResponse(data, status, cors) { return new Response(JSON.stringify(data), { status, headers: { ...cors, "Content-Type": "application/json; charset=utf-8", "Cache-Control": "no-store", "X-Content-Type-Options": "nosniff" } }); } function cleanText(value, maxLength = 20000) { if (typeof value !== "string") { return ""; } return value.trim().slice(0, maxLength); } async function callGroq(env, messages, options = {}) { if (!env.GROQ_API_KEY) { throw new Error("GROQ_API_KEY is not configured."); } const model = options.model || CHAT_MODEL; const response = await fetch( "https://api.groq.com/openai/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${env.GROQ_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model, messages, temperature: 0.2, max_tokens: 1800 }) } ); const raw = await response.text(); let data; try { data = JSON.parse(raw); } catch { throw new Error("The AI provider returned an invalid response."); } if (!response.ok) { console.error("Groq API error status:", response.status); const providerMessage = data?.error?.message || "AI provider request failed."; throw new Error( `AI provider error (${response.status}): ${providerMessage}` ); } const answer = data?.choices?.[0]?.message?.content; if (typeof answer !== "string" || !answer.trim()) { throw new Error("The AI provider returned an empty answer."); } return answer.trim(); } function makeSystemMessage(language) { const safeLanguage = cleanText(language, 40) || "Hindi"; return { role: "system", content: SYSTEM_PROMPT + "\n\nPreferred response language: " + safeLanguage + "." }; } function isValidImageDataUrl(value) { if (typeof value !== "string") return false; // Accept common browser-generated image data URLs. return /^data:image\/(png|jpeg|jpg|webp|gif);base64,[A-Za-z0-9+/=\r\n]+$/i .test(value); } function extractBase64(dataUrl) { const commaIndex = dataUrl.indexOf(","); if (commaIndex < 0) { throw new Error("Invalid image data URL."); } return dataUrl.slice(commaIndex + 1).replace(/\s/g, ""); } function getImageMimeType(dataUrl) { const match = dataUrl.match(/^data:(image\/[a-z0-9.+-]+);base64,/i); return match ? match[1].toLowerCase() : "image/jpeg"; } async function handleChat(request, env, cors) { let body; try { body = await request.json(); } catch { return jsonResponse( { error: "Request body must be valid JSON." }, 400, cors ); } const message = cleanText(body.message, 20000); const language = cleanText(body.language, 40) || "Hindi"; if (!message) { return jsonResponse( { error: "Please provide a message." }, 400, cors ); } const messages = [ makeSystemMessage(language), { role: "user", content: message } ]; try { const answer = await callGroq(env, messages); return jsonResponse( { answer, reply: answer, verified: false, verificationNote: "This answer was generated by AI. Live web verification was not performed." }, 200, cors ); } catch (error) { console.error("Chat error:", error.message); const status = error.message.includes("not configured") ? 503 : 502; return jsonResponse( { error: "Unable to generate an answer right now.", details: error.message }, status, cors ); } } async function handleVision(request, env, cors) { let body; try { body = await request.json(); } catch { return jsonResponse( { error: "Request body must be valid JSON." }, 400, cors ); } const image = body.image; const question = cleanText(body.question, 5000) || "Describe this image and answer any question visible in it."; const language = cleanText(body.language, 40) || "Hindi"; if (!isValidImageDataUrl(image)) { return jsonResponse( { error: "Invalid image. Send a supported image as a base64 data URL." }, 400, cors ); } // Keep request sizes reasonable. if (image.length > 8 * 1024 * 1024) { return jsonResponse( { error: "Image is too large. Please use a smaller image." }, 413, cors ); } const mimeType = getImageMimeType(image); const base64 = extractBase64(image); // Configure this secret to a currently supported Groq vision model. const visionModel = env.GROQ_VISION_MODEL || "meta-llama/llama-4-scout-17b-16e-instruct"; const messages = [ makeSystemMessage(language), { role: "user", content: [ { type: "text", text: question }, { type: "image_url", image_url: { url: `data:${mimeType};base64,${base64}` } } ] } ]; try { const answer = await callGroq(env, messages, { model: visionModel }); return jsonResponse( { answer, reply: answer, verified: false, verificationNote: "The image was analyzed by AI. External facts were not independently verified." }, 200, cors ); } catch (error) { console.error("Vision error:", error.message); return jsonResponse( { error: "Unable to analyze the image right now.", details: error.message }, 502, cors ); } } export default { async fetch(request, env) { const url = new URL(request.url); const origin = request.headers.get("Origin") || ""; const cors = corsHeaders(origin, env); if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } if (request.method !== "POST") { return jsonResponse( { error: "Method not allowed. Use POST." }, 405, cors ); } if (url.pathname === "/api/chat") { return handleChat(request, env, cors); } if (url.pathname === "/api/vision") { return handleVision(request, env, cors); } return jsonResponse( { error: "Endpoint not found." }, 404, cors ); } };
