// ========================================================== // CHAT // ========================================================== async function sendMessage( useSearch = false, customText = null ) { const text = customText !== null ? String(customText).trim() : input.value.trim(); if (!text) { setStatus( "कृपया पहले अपना सवाल लिखें।" ); return; } // ========================================================== // LOCAL DATE // तारीख के सवाल में Groq को request नहीं जाएगी // ========================================================== const asksDateOnly = /^(आज|आज की तारीख|आज का दिन|today|today's date|current date)\s*[?؟.!]*$/i.test( text ); if (asksDateOnly) { const localDate = new Intl.DateTimeFormat("hi-IN", { timeZone: "Asia/Kolkata", weekday: "long", day: "numeric", month: "long", year: "numeric", }).format(new Date()); addMessage( "user", text ); messages.push({ role: "user", content: text }); const dateAnswer = `आज की तारीख है ${localDate}।`; lastAnswer = dateAnswer; addMessage( "assistant", dateAnswer ); messages.push({ role: "assistant", content: dateAnswer }); input.value = ""; setStatus(""); return; } // ========================================================== // NORMAL MESSAGE // ========================================================== addMessage( "user", text ); messages.push({ role: "user", content: text }); input.value = ""; setStatus( useSearch ? "🌐 Internet Search से जानकारी खोजी जा रही है..." : "सारथी जवाब तैयार कर रहा है..." ); sendButton.disabled = true; searchButton.disabled = true; try { const response = await fetch( API_BASE + "/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: text, history: messages, webSearch: Boolean(useSearch) }) } ); let data = null; try { data = await response.json(); } catch { throw new Error( "Server response पढ़ा नहीं जा सका।" ); } if (!response.ok) { throw new Error( data?.answer || data?.error || data?.detail || "जवाब नहीं मिल पाया।" ); } const reply = data?.answer || "कोई जवाब नहीं मिला।"; lastAnswer = reply; addMessage( "assistant", reply ); messages.push({ role: "assistant", content: reply }); if ( data?.selfChecked === true ) { if (data?.searchUsed === true) { setStatus("✓ Internet Search से ताज़ा जानकारी खोजकर जवाब दिया गया"); } else if (data?.selfCorrected === true) { setStatus("✓ Self-check करके जवाब सुधारा गया"); } else if (data?.selfChecked === true) { setStatus("✓ Self-check करके जवाब जाँचा गया"); } else { setStatus(""); } } catch (error) { console.error( "Chat error:", error ); addMessage( "assistant", "अभी जवाब नहीं मिल पाया। कृपया फिर से कोशिश करें।" ); setStatus( "त्रुटि: " + ( error?.message || "Unknown error" ) ); } finally { sendButton.disabled = false; searchButton.disabled = false; input.focus(); } }
